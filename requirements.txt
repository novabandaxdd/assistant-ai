#  ║  A — NVIDIA CUDA (Linux ou Windows, GPU GeForce / RTX / Tesla)      ║
#  ╚══════════════════════════════════════════════════════════════════════╝
#
#   # CUDA 12.4 (padrão recomendado para drivers >= 525)
#   pip install torch torchvision torchaudio \
#       --index-url https://download.pytorch.org/whl/cu124
#
#   # CUDA 11.8 (drivers mais antigos)
#   pip install torch torchvision torchaudio \
#       --index-url https://download.pytorch.org/whl/cu118

#
#  ╔══════════════════════════════════════════════════════════════════════╗
#  ║  B — AMD ROCm / HIP (Linux, GPU RX 6700 XT / RX 7xxx / etc.)       ║
#  ╚══════════════════════════════════════════════════════════════════════╝
#
#   Requisito: ROCm 6.x instalado no sistema
#   Verifique: rocm-smi --version
#
#   # ROCm 6.2 (recomendado — inclui suporte à série RX 6000/7000)
#   pip install torch torchvision torchaudio \
#       --index-url https://download.pytorch.org/whl/rocm6.2
#
#   # Caso o modelo reporte "cpu" mesmo com ROCm instalado:
#   export HSA_OVERRIDE_GFX_VERSION=10.3.0   # para RX 6700 XT (gfx1031)
#   # Adicione essa variável ao ~/.bashrc ou ~/.profile para persistência.
#
#   Referência de GFX_VERSION por GPU AMD:
#     RX 6600 / 6700 XT / 6800 XT  → 10.3.0
#     RX 7900 XT / 7900 XTX         → 11.0.0
#     RX 7600 / 7700 / 7800 XT      → 11.0.2
#
#
#  ╔══════════════════════════════════════════════════════════════════════╗
#  ║  C — Apple Silicon M1 / M2 / M3 / M4 (macOS 12.3+)                 ║
#  ╚══════════════════════════════════════════════════════════════════════╝
#
#   # MPS (Metal Performance Shaders) — incluído no PyTorch padrão para macOS
#   pip install torch torchvision torchaudio
#
#   Requisitos:
#     macOS 12.3 (Monterey) ou superior
#     PyTorch >= 1.13 (MPS disponível a partir desta versão)

#  ╔══════════════════════════════════════════════════════════════════════╗
